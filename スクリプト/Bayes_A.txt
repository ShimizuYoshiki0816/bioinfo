library(BGLR)

n <- length(Y)
n_repeats <- 10
n_folds <- 10

pred_corrsAHC <- matrix(NA, nrow = n_repeats, ncol = n_folds)

for (r in 1:n_repeats) {
  cat("Repeat", r, "\n")
  
  # repeatごとにシード固定
  seed <- 100 + r
  set.seed(seed)
  
  # fold作成
  folds <- sample(rep(1:n_folds, length.out = n))
  y_hatAHC <- rep(NA, n)
  
  for (k in 1:n_folds) {
    cat("  Fold", k, "\n")
    
    test_idx <- which(folds == k)
    
    # 欠損化
    y_cv <- Y
    y_cv[test_idx] <- NA
    
    # BayesAモデル
    modelAHC <- BGLR(
      y = y_cv,
      ETA = list(list(X = X, model = "BayesA")),
      nIter = 10000,
      burnIn = 2000,
      verbose = FALSE
    )
    
    # 予測値
    y_hatAHC[test_idx] <- modelAHC$yHat[test_idx]
    
    # foldごとの精度
    pred_corrsAHC[r, k] <- cor(Y[test_idx],
                               y_hatAHC[test_idx],
                               use = "complete.obs")
  }
}

# 平均と標準誤差
mean_predAHC <- mean(pred_corrsAHC, na.rm = TRUE)
se_predAHC <- sd(pred_corrsAHC, na.rm = TRUE) / sqrt(n_repeats * n_folds)

cat("\n==== BayesA Predictive Ability ====\n")
cat("Mean Pearson's r:", round(mean_predAHC, 4), "\n")
cat("Standard Error  :", round(se_predAHC, 4), "\n")
