import pandas as pd
import numpy as np
from sklearn.svm import SVR
from sklearn.model_selection import KFold
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import StandardScaler
from scipy.stats import pearsonr
import math

# データ読み込みと前処理
pheno = pd.read_csv("~/HC_origin.csv")
geno = pd.read_csv("~/geno2.csv")
data = pd.concat([pheno[["HC"]], geno], axis=1).dropna()

y = data["HC"].values
X = data.drop(columns="HC").values

n_repeats, n_folds = 10, 5
corrs, mses = [], []

for repeat in range(n_repeats):
    seed = 100 + repeat 
    kf = KFold(n_splits=n_folds, shuffle=True, random_state=seed)

    for train_idx, test_idx in kf.split(X):
        X_train, X_test = X[train_idx], X[test_idx]
        y_train, y_test = y[train_idx], y[test_idx]

        scaler_X = StandardScaler()
        scaler_y = StandardScaler()

        X_train_scaled = scaler_X.fit_transform(X_train)
        X_test_scaled = scaler_X.transform(X_test)
        y_train_scaled = scaler_y.fit_transform(y_train.reshape(-1, 1)).ravel()

        model = SVR(kernel="poly", gamma="auto")
        model.fit(X_train_scaled, y_train_scaled)
        y_pred_scaled = model.predict(X_test_scaled)
        y_pred = scaler_y.inverse_transform(y_pred_scaled.reshape(-1, 1)).ravel()

        if np.std(y_test) == 0 or np.std(y_pred) == 0:
            continue

        corr = pearsonr(y_test, y_pred)[0]
        mse = mean_squared_error(y_test, y_pred)

        if not np.isnan(corr) and not np.isnan(mse):
            corrs.append(corr)
            mses.append(mse)

# 結果の表示
if corrs and mses:
    print("==== SVR (Polynomial Kernel) Predictive Performance ====")
    print(f"Mean Pearson's r : {np.mean(corrs):.4f} ± {np.std(corrs, ddof=1) / math.sqrt(len(corrs)):.4f}")
    print(f"Mean MSE         : {np.mean(mses):.4f} ± {np.std(mses, ddof=1) / math.sqrt(len(mses)):.4f}")
else:
    print("No valid folds to evaluate.")
