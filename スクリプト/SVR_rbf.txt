import pandas as pd
import numpy as np
from sklearn.svm import SVR
from sklearn.model_selection import KFold
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import StandardScaler

# データ読み込みと前処理
pheno = pd.read_csv("~/HC_origin.csv")
geno = pd.read_csv("~/geno2.csv")
data = pd.concat([pheno[["HC"]], geno], axis=1).dropna()

y = data["HC"].values
X = data.drop(columns="HC").values

n_repeats, n_folds = 10, 5
corrs, mses = [], []

for repeat in range(n_repeats):
    seed = 100 + repeat
    kf = KFold(n_splits=n_folds, shuffle=True, random_state=seed)

    for train_idx, test_idx in kf.split(X):
        X_train, X_test = X[train_idx], X[test_idx]
        y_train, y_test = y[train_idx], y[test_idx]

        scaler = StandardScaler()
        X_train_scaled = scaler.fit_transform(X_train)
        X_test_scaled = scaler.transform(X_test)

        model = SVR(kernel="rbf", gamma="auto")
        model.fit(X_train_scaled, y_train)
        y_pred = model.predict(X_test_scaled)

        if np.std(y_test) == 0 or np.std(y_pred) == 0:
            continue

        corr = np.corrcoef(y_test, y_pred)[0, 1]
        mse = mean_squared_error(y_test, y_pred)

        if not np.isnan(corr) and not np.isnan(mse):
            corrs.append(corr)
            mses.append(mse)

# 結果出力
if corrs and mses:
    print("==== SVR (RBF Kernel, Scaled X) Predictive Performance ====")
    print(f"Mean Pearson's r : {np.mean(corrs):.4f} ± {np.std(corrs, ddof=1) / np.sqrt(len(corrs)):.4f}")
    print(f"Mean MSE         : {np.mean(mses):.4f} ± {np.std(mses, ddof=1) / np.sqrt(len(mses)):.4f}")
else:
    print("No valid folds to evaluate.")
